```python
    tenant_id: str,
    kb_id: str,
    doc_id: str,
    subgraph: nx.Graph,
    embedding_model,
    callback,
):

    if old_graph is not None:

        new_graph.nodes[node_name]["pagerank"] = pagerank

    now = asyncio.get_running_loop().time()
    callback(msg=f"merging subgraph for doc {doc_id} into the global graph done in {now - start:.2f} seconds.")
    return new_graph
```

### 2.3 图合并算法

**函数**: `graph_merge()` ([utils.py:306-336](file:///d:/work/rag/ragflow/rag/graphrag/utils.py#L306-L336))

```python
def graph_merge(g1: nx.Graph, g2: nx.Graph, change: GraphChange):
    """将 g2 合并到 g1（原地修改）"""
    # (1) 节点合并
    for node_name, attr in g2.nodes(data=True):
        change.added_updated_nodes.add(node_name)
        if not g1.has_node(node_name):
            g1.add_node(node_name, **attr)
            continue
        node = g1.nodes[node_name]
        # 合并描述: 用 <SEP> 拼接多个文档的描述
        node["description"] += GRAPH_FIELD_SEP + attr["description"]
        # 合并来源文档列表
        node["source_id"] += attr["source_id"]
    
    # (2) 边合并
    for source, target, attr in g2.edges(data=True):
        change.added_updated_edges.add(get_from_to(source, target))
        edge = g1.get_edge_data(source, target)
        if edge is None:
            g1.add_edge(source, target, **attr)
            continue
        # 累加权重
        edge["weight"] += attr.get("weight", 0)
        # 合并描述
        edge["description"] += GRAPH_FIELD_SEP + attr["description"]
        # 合并关键词
        edge["keywords"] += attr["keywords"]
        # 合并来源文档列表
        edge["source_id"] += attr["source_id"]
    
    # (3) 更新节点度数
    for node_degree in g1.degree:
        g1.nodes[str(node_degree[0])]["rank"] = int(node_degree[1])
    
    # (4) 更新图级别元数据
    if "source_id" not in g1.graph:
        g1.graph["source_id"] = []
    g1.graph["source_id"] += g2.graph.get("source_id", [])
    return g1
```

**合并策略**:
- **节点**: 同名实体的描述用 `<SEP>` 拼接（不去重，保留所有文档的描述片段）
- **边**: 同端点关系的权重累加、描述拼接、关键词列表合并
- **source_id**: 追加所有贡献文档的 ID

```python
resolution_pending = with_resolution and not has_phase_marker(kb_id, PHASE_RESOLUTION)
community_pending = with_community and not has_phase_marker(kb_id, PHASE_COMMUNITY)

# ... 执行子图合并 ...

# 合并后清除标记（因为图发生变化）
clear_phase_markers(kb_id)
resolution_pending = with_resolution
community_pending = with_community

# 执行 Resolution
if resolution_pending:
    await resolve_entities(...)
    set_phase_marker(kb_id, PHASE_RESOLUTION)
elif with_resolution:
    callback(msg=f"[GraphRAG] dataset:{kb_id} entity resolution already completed previously, skipping.")

# 执行 Community Detection
if community_pending:
    await extract_community(...)
    set_phase_marker(kb_id, PHASE_COMMUNITY)
elif with_community:
    callback(msg=f"[GraphRAG] dataset:{kb_id} community detection already completed previously, skipping.")
```

**逻辑流程**:
1. 新文档合并后 → 清除所有标记 → Resolution 和 Community 重新执行
2. 仅查询/检索操作 → 标记存在 → 跳过重复计算
3. 执行完成后 → 设置标记 → 下次任务直接跳过

## 八、性能优化点

1. **批量 Embedding 预热**: 将 N 次单独 API 调用合并为 ceil(N/100) 次批量调用
2. **并发子图生成**: 使用 `asyncio.gather` 并发处理多个文档
3. **顺序合并**: 避免并发合并导致的图冲突，使用分布式锁保护
4. **阶段标记**: 避免重复执行 Resolution 和 Community Detection
5. **增量更新**: 只处理 `change.added_updated_nodes/edges`，不重复处理全图
6. **缓存机制**: Redis 缓存 Embedding 结果，24 小时 TTL
7. **原子性持久化**: 先生成所有新数据，再原子替换旧数据，保证崩溃恢复能力
